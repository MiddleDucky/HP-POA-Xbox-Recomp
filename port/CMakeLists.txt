cmake_minimum_required(VERSION 3.20)

# ============================================================================
# hppoa - Static Recompilation
#
# Copy this template to your game project root and customize:
#   1. Change the project name below
#   2. Set XBOXRECOMP_DIR to where you cloned xboxrecomp
#   3. Adjust source file lists as you add game-specific code
#   4. Add any game-specific compile definitions or link libraries
# ============================================================================

project(hppoa C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Common definitions
add_compile_definitions(
    WIN32_LEAN_AND_MEAN
    NOMINMAX
)

# ── xboxrecomp toolkit ──────────────────────────────────────────
#
# Option A: xboxrecomp as a subdirectory (simplest)
#   Place xboxrecomp/ next to your game project and use a relative path.
#
# Option B: find_package (if you install xboxrecomp system-wide)
#   find_package(xboxrecomp REQUIRED)
#
# Adjust this path to point to your xboxrecomp clone:
set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH
    "Path to the xboxrecomp toolkit root directory")

# Add xboxrecomp as a subdirectory -- this builds all 6 runtime libraries
# and provides the 'xboxrecomp' umbrella target to link against.
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# ── Recompiled game executable ──────────────────────────────────

# Collect generated recompiled code.
# The recomp pipeline outputs files into src/recomp/gen/.
# With --split 1000, you get gen_0000.c, gen_0001.c, etc.
# CONFIGURE_DEPENDS so re-running the recompiler picks up added/removed files
# without a manual re-configure. A plain GLOB is evaluated once at configure
# time, so a newly generated file (an extra chunk, recomp_stubs_unresolved.c)
# is silently left out of the build and shows up as unresolved externals.
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS "src/recomp/gen/*.c")

# Core game sources
set(GAME_SOURCES
    src/main.c
    src/recomp_manual.c
    src/recomp_extra.c
    # DirectSound on a host mixer (XAudio2) instead of the emulated APU.
    src/hle_dsound.c
    # The renderer's per-title questions (menu movie, loading screen, text icons): none found yet.
    src/hle_video.c
    # XInput straight from the PC pad (SCDA style), rumble on the host.
    src/hle_input.c
    # Debug traces: RECOMP_CLOCK_TRACE ([CLOCK], the game clock).
    src/hle_debug.c
    # HPPOA_CALL_PROFILE: each F7 window's calls and object snapshots.
    src/hle_deeptrace.c
    # 60 fps: the game clock's minimum step (frame_rate).
    src/hle_clock.c
    # The diagnostics report's game side (hppoa_diag.exe, Ctrl+Shift+R): state, stacks, a burst of frame captures.
    src/hle_diag.c
    # True 16:9: the screen aspect the game's cameras use.
    src/hle_widescreen.c
    # Every room beyond a doorway the camera faces drawn, not only doorways on screen: RECOMP_DRAW_ALL_ROOMS.
    src/hle_portals.c
    # The game's asynchronous file reads (level packages): RECOMP_LOADER_TRACE.
    src/hle_loader.c
    # Quit To Desktop: Back on the front end and the pause menu, an are-you-sure box (menu.cpp draws it).
    src/hle_quit.c
    # The trainer's game side (F6, renderer/app/trainer.cpp): the title's tuning variables, its debug menu, health,
    # items, game speed.
    src/hle_trainer.c
    src/profiler.c
    # Direct3D 8 on a host D3D12 renderer, ported from SCDA-XBERecomp
    # (src/renderer, MIT): the XDK's D3D entry points instead of the NV2A.
    src/renderer/xdk/hle_d3d8.cpp
    # Hand-written over generated bodies: the recompile (regen.sh, run from hppoa/) must pass
    #   --exclude-manual port/src/hle_dsound.c
    #   --exclude-manual port/src/hle_input.c
    #   --exclude-manual port/src/hle_debug.c
    #   --exclude-manual port/src/hle_clock.c
    #   --exclude-manual port/src/hle_widescreen.c
    #   --exclude-manual port/src/hle_portals.c
    #   --exclude-manual port/src/hle_loader.c
    #   --exclude-manual port/src/hle_quit.c
    #   --exclude-manual port/src/hle_trainer.c
    #   --exclude-manual port/src/renderer/xdk/hle_d3d8.cpp
)

# The recomp pipeline also generates a dispatch header and function declarations.
# These are included by the generated .c files:
#   src/recomp/gen/recomp_funcs.h    - extern declarations for all recompiled functions
#   src/recomp/recomp_dispatch.h     - binary search dispatch table
#   src/recomp/recomp_types.h        - register model, ICALL macros, MEM helpers

# resource 1: the port's own icon (res/app.ico, tools/make_icon.py poa: a wand drawn from shapes in the launcher's
# purple, no game art), which fb_present.c and the launcher load for their windows
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/hppoa_res.rc"
     CONTENT "// generated by CMakeLists.txt\n1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/res/app.ico\"\n")
# the icon is named by these scripts, not part of them: without this a new res/app.ico left the old one in the exes
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/hppoa_res.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/res/app.ico")
add_executable(${PROJECT_NAME} WIN32
    ${CMAKE_CURRENT_BINARY_DIR}/hppoa_res.rc
    ${GAME_SOURCES}
    ${RECOMP_GEN_SOURCES}
)

# the run log's first line (main.c): the kind of build and the repository's commit, so a player's log says which
# build it came from (no release version yet). build_info.h is remade on every build (tools/build_info.cmake)
add_custom_target(hppoa_build_info
    COMMAND ${CMAKE_COMMAND} -DREPO=${CMAKE_CURRENT_SOURCE_DIR}/.. -DOUT=${CMAKE_CURRENT_BINARY_DIR}/build_info.h
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/build_info.cmake
    BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/build_info.h
    VERBATIM)
add_dependencies(${PROJECT_NAME} hppoa_build_info)

# HPPOA_RELEASE: the build players get (tools/package_release.sh): F6 (the trainer) and F12 / Insert (frame
# captures) do nothing; the run log names the version. Everything else is as in a dev build
option(HPPOA_RELEASE "Build for players: no F6 trainer or F12 captures" OFF)
set(HPPOA_VERSION "0.1.0" CACHE STRING "Release version, shown in the run log")
if(HPPOA_RELEASE)
    target_compile_definitions(${PROJECT_NAME} PRIVATE HPPOA_RELEASE=1)
    set(HPPOA_KIND "release v${HPPOA_VERSION}")
else()
    set(HPPOA_KIND "dev ${CMAKE_BUILD_TYPE}")
endif()
set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/src/main.c PROPERTIES
    COMPILE_DEFINITIONS "HPPOA_BUILD_KIND=\"${HPPOA_KIND}\""
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}")

# MinGW GCC 16 at -O3 sometimes hoists the address of a thread local array out of a loop as a plain global: the
# lifted code then reads the TLS template instead of the thread's copy (g_fp_stack in sub_00036EC0, which turned the
# Whomping Willow's right hand to angle 0). -O2 keeps it out; the post build check below fails on any such access.
if(WIN32)
    set_source_files_properties(${RECOMP_GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-O2")
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_tls_refs.py
                    --objdump ${CMAKE_OBJDUMP} --nm ${CMAKE_NM} $<TARGET_FILE:${PROJECT_NAME}>
            VERBATIM)
    endif()
endif()

# Link against the xboxrecomp umbrella target (all 6 runtime libraries)
target_link_libraries(${PROJECT_NAME} PRIVATE xboxrecomp)

# Include paths for generated recomp headers
target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp
    ${XBOXRECOMP_DIR}/include
)

# SCDA-XBERecomp's renderer (src/renderer/gpu): XDK Direct3D state to D3D12.
if(WIN32)
    target_sources(${PROJECT_NAME} PRIVATE
        src/renderer/app/app.cpp
        src/renderer/gpu/d3d12.cpp
        src/renderer/gpu/pipeline_cache.cpp
        src/renderer/gpu/render_thread.cpp
        src/renderer/gpu/shaders.cpp
        src/renderer/gpu/texture.cpp
        src/renderer/gpu/vertex.cpp
        src/renderer/app/menu.cpp
        src/renderer/app/input.cpp
        src/renderer/app/trainer.cpp
    )
    # Dear ImGui v1.91.0 (third_party/imgui, MIT), as SCDA: the F10 options panel
    set(IMGUI_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
    target_sources(${PROJECT_NAME} PRIVATE
        ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp
        ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp
        ${IMGUI_DIR}/backends/imgui_impl_dx12.cpp)
    set_source_files_properties(
        ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp
        ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp
        ${IMGUI_DIR}/backends/imgui_impl_dx12.cpp
        PROPERTIES INCLUDE_DIRECTORIES "${IMGUI_DIR}" COMPILE_OPTIONS "-O2")
    set_source_files_properties(
        src/renderer/xdk/hle_d3d8.cpp src/renderer/app/app.cpp src/renderer/gpu/d3d12.cpp
        src/renderer/gpu/pipeline_cache.cpp src/renderer/gpu/render_thread.cpp src/renderer/gpu/shaders.cpp
        src/renderer/gpu/texture.cpp src/renderer/gpu/vertex.cpp src/renderer/app/menu.cpp src/renderer/app/input.cpp
        src/renderer/app/trainer.cpp
        PROPERTIES INCLUDE_DIRECTORIES
        "${CMAKE_CURRENT_SOURCE_DIR}/src/renderer;${XBOXRECOMP_DIR}/src;${IMGUI_DIR}"
        COMPILE_OPTIONS "-msse4.1")
    set_target_properties(${PROJECT_NAME} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
endif()

if(WIN32)
    # the C++ renderer must not need MinGW's libstdc++/libgcc/winpthread DLLs at run time
    target_link_options(hppoa PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_libraries(hppoa PRIVATE d3d11 d3d12 d3dcompiler dxgi dxguid xinput1_4 winmm dbghelp xaudio2_8 ole32 imm32 dwmapi gdi32)
else()
    target_link_options(hppoa PRIVATE -rdynamic)
endif()

# MSVC-specific: large address aware, increase section limit for big generated files
if(MSVC)
    target_link_options(${PROJECT_NAME} PRIVATE
        /LARGEADDRESSAWARE
    )
    # The generated code files can be very large. Increase object file section limit.
    target_compile_options(${PROJECT_NAME} PRIVATE /bigobj)
endif()
